#### 2.1 使用 useReducer 集中管理 UI 状态

```typescript
// 定义状态类型
type UIState = {
  editMode: boolean;
  storyForm: { title: string; premise: string };
  editingPanel: Partial<Panel> | null;
  isDialogOpen: boolean;
  deletingPanelId: string | null;
};

type UIAction =
  | { type: 'SET_EDIT_MODE'; payload: boolean }
  | { type: 'UPDATE_STORY_FORM'; payload: Partial<UIState['storyForm']> }
  | { type: 'SET_EDITING_PANEL'; payload: Partial<Panel> | null }
  | { type: 'SET_DIALOG_OPEN'; payload: boolean }
  | { type: 'SET_DELETING_PANEL'; payload: string | null }
  | { type: 'RESET_FORM' };

function uiReducer(state: UIState, action: UIAction): UIState {
  switch (action.type) {
    case 'SET_EDIT_MODE':
      return { ...state, editMode: action.payload };
    case 'UPDATE_STORY_FORM':
      return { 
        ...state, 
        storyForm: { ...state.storyForm, ...action.payload } 
      };
    case 'SET_EDITING_PANEL':
      return { ...state, editingPanel: action.payload };
    case 'SET_DIALOG_OPEN':
      return { ...state, isDialogOpen: action.payload };
    case 'SET_DELETING_PANEL':
      return { ...state, deletingPanelId: action.payload };
    case 'RESET_FORM':
      return {
        ...state,
        editMode: false,
        editingPanel: null,
        isDialogOpen: false,
        deletingPanelId: null,
      };
    default:
      return state;
  }
}

// 在组件中使用
const [uiState, dispatch] = useReducer(uiReducer, {
  editMode: false,
  storyForm: { title: "", premise: "" },
  editingPanel: null,
  isDialogOpen: false,
  deletingPanelId: null,
});
```

**收益**:

// 创建自定义 Hook 持久化编辑状态
function usePersistedEditState(storyId: string) {
  const key = `story-edit-${storyId}`;
  
  const [state, setState] = useState<Partial<Panel> | null>(() => {
    if (typeof window === 'undefined') return null;
    const saved = localStorage.getItem(key);
    return saved ? JSON.parse(saved) : null;
  });

  useEffect(() => {
    if (state) {
      localStorage.setItem(key, JSON.stringify(state));
    } else {
      localStorage.removeItem(key);
    }
  }, [key, state]);

  const clearState = useCallback(() => {
    setState(null);
    localStorage.removeItem(key);
  }, [key]);

  return [state, setState, clearState] as const;
}

// 优化后的图片生成 Hook
export function usePanelGenerationOptimized({
  onSuccess,
  onError,
}: UsePanelGenerationProps = {}) {
  const [generatingPanels, setGeneratingPanels] = useState<
    Map<string, GenerationTask>
  >(new Map());

  // 智能轮询:根据状态调整轮询间隔
  const getPollingInterval = (status: GenerationStatus): number => {
    switch (status) {
      case 'pending':
        return 5000; // 等待中,5秒轮询
      case 'processing':
        return 2000; // 处理中,2秒轮询
      default:
        return 0; // 完成或失败,停止轮询
    }
  };

  const startPolling = useCallback(
    (panelId: string, taskId: string) => {
      const poll = async () => {
        try {
          const response = await fetch(
            `/api/panel/generate-image/status?task_id=${taskId}`
          );

          if (!response.ok) {
            throw new Error("Failed to fetch status");
          }

          const status: TaskStatusResponse = await response.json();

          setGeneratingPanels((prev) => {
            const newMap = new Map(prev);
            const existing = newMap.get(panelId);
            if (existing) {
              newMap.set(panelId, {
                ...existing,
                status: status.status,
                progress: status.progress,
                imageUrl: status.imageUrl,
                error: status.error,
              });
            }
            return newMap;
          });

          // 根据状态决定下次轮询时间
          const interval = getPollingInterval(status.status);
          if (interval > 0) {
            setTimeout(() => poll(), interval);
          } else {
            // 任务完成
            if (status.status === "completed" && status.imageUrl) {
              onSuccess?.(panelId, status.imageUrl);
            } else if (status.status === "failed") {
              onError?.(panelId, status.error || "生成失败");
            }
          }
        } catch (error) {
          console.error("Polling error:", error);
          // 失败后延长间隔重试
          setTimeout(() => poll(), 10000);
        }
      };

      poll();
    },
    [onSuccess, onError]
  );

  // ... 其他方法
}

#### 3.2 添加请求去重与智能重试

```typescript
// 创建请求去重 Hook
function useDedupedRequest<T>(
  requestFn: () => Promise<T>,
  key: string
) {
  const requestCacheRef = useRef<Map<string, Promise<T>>>(new Map());

  return useCallback(async (): Promise<T> => {
    const cached = requestCacheRef.current.get(key);
    if (cached) {
      return cached;
    }

    const promise = requestFn();
    requestCacheRef.current.set(key, promise);

    try {
      const result = await promise;
      return result;
    } finally {
      // 请求完成后清除缓存
      requestCacheRef.current.delete(key);
    }
  }, [requestFn, key]);
}

// 智能重试
async function fetchWithRetry(
  url: string,
  options: RequestInit,
  maxRetries = 3
) {
  let lastError: Error | null = null;
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options);
      if (response.ok) return response;
      
      // 5xx 错误重试,4xx 错误不重试
      if (response.status >= 500) {
        lastError = new Error(`Server error: ${response.status}`);
        await new Promise(resolve => 
          setTimeout(resolve, Math.pow(2, i) * 1000)
        ); // 指数退避
        continue;
      }
      
      return response;
    } catch (error) {
      lastError = error as Error;
      if (i < maxRetries - 1) {
        await new Promise(resolve => 
          setTimeout(resolve, Math.pow(2, i) * 1000)
        );
      }
    }
  }
  
  throw lastError || new Error('Request failed');
}
```

**收益**:
- 避免重复请求
- 网络波动时自动重试
- 用户体验更稳定

// ErrorBoundary.tsx
'use client';

import { Component, ReactNode } from 'react';
import { Button } from '@/components/ui/button';
import { AlertTriangle } from 'lucide-react';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('Error caught by boundary:', error, errorInfo);
    // 可以上报到错误监控服务
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="flex flex-col items-center justify-center min-h-[50vh] space-y-4">
          <AlertTriangle className="w-16 h-16 text-destructive" />
          <h2 className="text-2xl font-bold">页面出现错误</h2>
          <p className="text-muted-foreground">
            {this.state.error?.message || '未知错误'}
          </p>
          <Button
            onClick={() => {
              this.setState({ hasError: false, error: null });
              window.location.reload();
            }}
          >
            重新加载
          </Button>
        </div>
      );
    }

    return this.props.children;
  }
}

// 在 page.tsx 中包裹

// 创建详细的错误处理工具
function getErrorMessage(error: unknown): string {
  if (error instanceof Error) {
    return error.message;
  }
  if (typeof error === 'string') {
    return error;
  }
  return '未知错误';
}

function getErrorType(error: unknown): 'network' | 'validation' | 'server' | 'unknown' {
  if (error instanceof TypeError && error.message.includes('fetch')) {
    return 'network';
  }
  // 根据错误信息判断类型
  const message = getErrorMessage(error);
  if (message.includes('验证') || message.includes('格式')) {
    return 'validation';
  }
  if (message.includes('服务器') || message.includes('500')) {
    return 'server';
  }
  return 'unknown';
}

          },
        };

    },
    onSuccess: (data) => {

        return {
          ...old,
          story: { ...data.story, _optimistic: false },
        };
      });
    },
    onError: (err, variables, context) => {
      if (context?.previousData) {
        queryClient.setQueryData(['story', storyId], context.previousData);
      }
    },
  });
